**Diagram/distance core pattern** (cell 30, verbatim — promote this, do not reimplement):
```python
def _finite_pairs(dgm):
    """Filter H0's one infinite death; persim's distances cannot consume +inf."""
    d = np.asarray(dgm, dtype=float).reshape(-1, 2)
    if d.size == 0:
        return np.zeros((0, 2))
    return np.ascontiguousarray(d[np.isfinite(d[:, 1])])

def persistence_diagram(D: np.ndarray, maxdim: int) -> list[np.ndarray]:
    """D: (n, n) square symmetric distance matrix. Returns [dgm_H0, dgm_H1, ...], RAW
    (unnormalized) units -- D-05's normalization is a caller-side division, not baked in
    here, so the same raw diagram serves both the reference-max-persistence normalization
    (D-05) and any future re-normalization convention without recomputing ripser."""
    D = np.asarray(D, dtype=float)
    dgms = ripser(D, maxdim=maxdim, distance_matrix=True)["dgms"]
    return [_finite_pairs(g) for g in dgms]

def bottleneck(model_dgm, ref_dgm) -> float:
    return float(persim.bottleneck(model_dgm, ref_dgm))

def wasserstein(model_dgm, ref_dgm) -> float:
    return float(persim.wasserstein(model_dgm, ref_dgm))

def saturation(ref_dgm) -> float:
    """Bottleneck value a model diagram lands on once nothing matches the reference's
    longest bar -- exactly half that bar's own life. A distance sitting here carries no
    further information."""
    L = ref_dgm[:, 1] - ref_dgm[:, 0]
    top = float(L.max()) if L.size else 0.0
    return 0.5 * top

def is_saturated(value, ref_dgm, tol=1e-9) -> bool:
    return abs(value - saturation(ref_dgm)) <= tol
```

**Deviation from the quick task the new module MUST make (not a copy-verbatim case):** the
quick task normalizes by **diameter** (its own §13 rule, cell 30's docstring). D-05 ratifies a
**different** rule: normalize by **the reference diagram's own max persistence**. Do not port
the diameter-normalization convention — write:
```python
def max_persistence(dgm: np.ndarray) -> float:
    """The reference diagram's own scale, per CONTEXT.md D-05. Empty diagram -> 0.0; the
    caller must guard division (near-zero for a contractible reference's H1 -- see
    RESEARCH.md Pitfall 2, the intrinsic-plane H1 denominator hazard)."""
    if dgm.size == 0:
        return 0.0
    return float((dgm[:, 1] - dgm[:, 0]).max())
```
